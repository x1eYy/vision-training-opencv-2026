cmake_minimum_required(VERSION 3.16)
project(vision_training LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenCV REQUIRED)
find_package(Eigen3 REQUIRED NO_MODULE)
add_library(vision_common src/common/common.cpp)
target_include_directories(vision_common PUBLIC include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(vision_common PUBLIC ${OpenCV_LIBS})
add_executable(task1_image src/task1_image/main.cpp)
target_link_libraries(task1_image PRIVATE vision_common)
add_executable(task2_fit src/task2_fit/main.cpp)
target_link_libraries(task2_fit PRIVATE vision_common Eigen3::Eigen)
add_executable(task3_windmill src/task3_windmill/main.cpp)
target_link_libraries(task3_windmill PRIVATE vision_common)
