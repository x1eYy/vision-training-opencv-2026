cmake_minimum_required(VERSION 3.16)
project(vision_training LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenCV REQUIRED)
find_package(Eigen3 REQUIRED NO_MODULE)
add_library(vision_common src/common/common.cpp)
target_include_directories(vision_common PUBLIC include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(vision_common PUBLIC ${OpenCV_LIBS})
add_executable(task1_image src/task1_image/main.cpp)
target_link_libraries(task1_image PRIVATE vision_common)
add_executable(task2_fit src/task2_fit/main.cpp)
target_link_libraries(task2_fit PRIVATE vision_common Eigen3::Eigen)
add_library(windmill_core src/task3_windmill/detector.cpp src/task3_windmill/tracker.cpp src/task3_windmill/render.cpp)
target_link_libraries(windmill_core PUBLIC vision_common)
add_executable(task3_windmill src/task3_windmill/main.cpp)
target_link_libraries(task3_windmill PRIVATE windmill_core)
add_executable(windmill_reference_check src/task3_windmill/reference_check.cpp)
target_link_libraries(windmill_reference_check PRIVATE windmill_core)
enable_testing()
add_executable(windmill_tracker_test src/task3_windmill/tracker_test.cpp)
target_link_libraries(windmill_tracker_test PRIVATE windmill_core)
add_test(NAME windmill_tracker_state_machine COMMAND windmill_tracker_test)
